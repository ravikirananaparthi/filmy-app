01 / Shared Motion State

Give the tab bar and every feed the same scroll-direction signal.

export default function TabLayout() {
  return (
    <MotionifyProvider threshold={10} supportIdle={false}>
      <Tabs
        tabBar={(props) => (
          <CollapsingTabBar {...props} />
        )}
      />
    </MotionifyProvider>
  );
}

// One provider. One direction source. No prop-drilling.

04 / Active Tab Spring

Animate selection separately from scroll visibility.

const ROUTE_SLOT = {
  index: 0,
  search: 1,
  favorites: 3,
  menu: 4,
};

function getIndicatorX(routeName: string) {
  const slot = ROUTE_SLOT[routeName];
  if (slot === undefined) return null;
  return slot * TAB_ITEM_WIDTH + INDICATOR_PADDING;
}
